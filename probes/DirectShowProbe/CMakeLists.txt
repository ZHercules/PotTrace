cmake_minimum_required(VERSION 3.24)
project(PotTraceDirectShowProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
get_filename_component(BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../out/deps/directshow-baseclasses" ABSOLUTE)
if(NOT EXISTS "${BASE_DIR}/streams.h")
 message(FATAL_ERROR "Download pinned Microsoft baseclasses into out/deps/directshow-baseclasses first")
endif()
file(GLOB BASE_SOURCES "${BASE_DIR}/*.cpp")
add_library(DirectShowBase STATIC ${BASE_SOURCES})
target_include_directories(DirectShowBase PUBLIC "${BASE_DIR}")
target_compile_definitions(DirectShowBase PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN)
target_compile_options(DirectShowBase PRIVATE /W0 /permissive /Zc:strictStrings-)
add_library(PotTraceGraphProbe SHARED GraphProbe.cpp GraphProbe.def)
target_compile_definitions(PotTraceGraphProbe PRIVATE NOMINMAX)
target_compile_options(PotTraceGraphProbe PRIVATE /W4 /utf-8)
target_link_libraries(PotTraceGraphProbe PRIVATE DirectShowBase strmiids winmm ole32 oleaut32 uuid advapi32 mfuuid)
set_target_properties(PotTraceGraphProbe PROPERTIES SUFFIX ".ax")
