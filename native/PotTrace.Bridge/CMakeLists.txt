cmake_minimum_required(VERSION 3.24)
project(PotTraceBridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(PotTraceBridge SHARED src/Bridge.cpp)
target_compile_definitions(PotTraceBridge PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(PotTraceBridge PRIVATE advapi32 ole32)
target_compile_options(PotTraceBridge PRIVATE /W4 /utf-8)
set_target_properties(PotTraceBridge PROPERTIES OUTPUT_NAME "PotTrace.Bridge")
enable_testing()
add_executable(BridgeTests ../../tests/native/BridgeTests.cpp)
target_include_directories(BridgeTests PRIVATE include)
target_compile_options(BridgeTests PRIVATE /W4 /utf-8)
add_test(NAME BridgeBehavior COMMAND BridgeTests)
add_executable(CallbackTests ../../tests/native/CallbackTests.cpp)
target_compile_definitions(CallbackTests PRIVATE POTTRACE_BRIDGE_TEST UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(CallbackTests PRIVATE /W4 /utf-8)
target_link_libraries(CallbackTests PRIVATE advapi32 ole32)
add_test(NAME CallbackState COMMAND CallbackTests)
add_executable(ObservationFenceTests ../../tests/native/ObservationFenceTests.cpp)
target_include_directories(ObservationFenceTests PRIVATE include)
target_compile_options(ObservationFenceTests PRIVATE /W4 /utf-8)
add_test(NAME ObservationFence COMMAND ObservationFenceTests)
add_executable(GraphTimelineTests ../../tests/native/GraphTimelineTests.cpp)
target_include_directories(GraphTimelineTests PRIVATE include)
target_compile_options(GraphTimelineTests PRIVATE /W4 /utf-8)
add_test(NAME GraphTimeline COMMAND GraphTimelineTests)
add_executable(MpscQueueTests ../../tests/native/MpscQueueTests.cpp)
target_include_directories(MpscQueueTests PRIVATE include)
target_compile_options(MpscQueueTests PRIVATE /W4 /utf-8)
add_test(NAME MpscQueue COMMAND MpscQueueTests)
add_executable(EffectiveRateTests ../../tests/native/EffectiveRateTests.cpp)
target_include_directories(EffectiveRateTests PRIVATE include)
target_compile_options(EffectiveRateTests PRIVATE /W4 /utf-8)
add_test(NAME EffectiveRate COMMAND EffectiveRateTests)
