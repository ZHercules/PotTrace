cmake_minimum_required(VERSION 3.24)
project(PotTraceCollector LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
get_filename_component(BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../out/deps/directshow-baseclasses" ABSOLUTE)
if(NOT EXISTS "${BASE_DIR}/streams.h")
 message(FATAL_ERROR "Pinned DirectShow baseclasses are required in out/deps/directshow-baseclasses")
endif()
file(GLOB BASE_SOURCES "${BASE_DIR}/*.cpp")
add_library(CollectorDirectShowBase STATIC ${BASE_SOURCES})
target_include_directories(CollectorDirectShowBase PUBLIC "${BASE_DIR}")
target_compile_definitions(CollectorDirectShowBase PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN)
target_compile_options(CollectorDirectShowBase PRIVATE /W0 /permissive /Zc:strictStrings-)
add_library(PotTraceCollector SHARED src/CollectorFilter.cpp Collector.def)
target_include_directories(PotTraceCollector PRIVATE include ../PotTrace.Bridge/include)
target_compile_definitions(PotTraceCollector PRIVATE NOMINMAX)
target_compile_options(PotTraceCollector PRIVATE /W4 /utf-8 /permissive /Zc:strictStrings-)
target_link_libraries(PotTraceCollector PRIVATE CollectorDirectShowBase strmiids winmm ole32 oleaut32 uuid advapi32 mfuuid)
set_target_properties(PotTraceCollector PROPERTIES OUTPUT_NAME "PotTrace.Collector" SUFFIX ".ax")
